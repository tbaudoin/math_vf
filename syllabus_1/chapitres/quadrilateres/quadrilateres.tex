    \begin{enumerate}[itemsep=1cm]
        \item Un quadrilatère peut posséder exactement trois angles droits.
        \item La somme de deux angles opposés d'un rectangle vaut $\ang{180}$.
        \item Dans un losange, les diagonales se coupent en formant quatre angles droits.
        \item Dans un trapèze, la somme des quatre angles est toujours égale à $\ang{360}$.
        \item Deux angles consécutifs d'un parallélogramme sont toujours égaux.
    \end{enumerate}
\end{exercise}

\section{Constructions de quadrilatères}
Réalise les constructions demandées sur une feuille annexe afin d'avoir tout
l'espace nécessaire.
\begin{enumerate}[itemsep=1em]
    \item Construis un carré $ABCD$ de \qty{5}{cm} de côté.
    
    \item Construis un rectangle $EFGH$ tel que $\longueur{EF} = \qty{7}{cm}$ et $\longueur{FG} = \qty{4}{cm}$.
    
    \item Construis un losange $IJKL$ dont les côtés mesurent \qty{5}{cm} et dont l'angle en $I$ mesure \ang{60}.
    
    \item Construis un parallélogramme $MNOP$ tel que $\longueur{MN} = \qty{6}{cm}$, $\longueur{NO} = \qty{4}{cm}$ et \\$\amp{MNO} = \ang{50}$.
    
    \item Construis un quadrilatère $ABCD$ dont les diagonales mesurent \qty{8}{cm} et \qty{6}{cm}, se coupent en leur milieu et sont perpendiculaires. Quelle est la nature de ce quadrilatère ?
    
    \item Construis un trapèze $RSTU$ tel que $RS \varparallel TU$,
    $\longueur{RS} = \qty{5}{cm}$, $\longueur{RU} = \qty{4}{cm}$ et
    $\longueur{TU} = \qty{3}{cm}$.
    
    \item Construis un triangle $ABC$ tel que $\longueur{AB} = \qty{6}{cm}$, $\longueur{AC} = \qty{5}{cm}$ et $\longueur{BC} = \qty{4}{cm}$. Place le point $D$ symétrique de $A$ par rapport au milieu de $[BC]$. Quelle est la nature du quadrilatère $ABCD$ ?
    
    \item Construis un quadrilatère $ABCD$ tel que $\longueur{AB} = \qty{5}{cm}$, $\longueur{BC} = \qty{6}{cm}$, $\widehat{ABC} = \ang{70}$, $\widehat{BCD} = \ang{110}$ et $\longueur{CD} = \qty{5}{cm}$.
    
    \item Construis un quadrilatère $ABCD$ dont les diagonales se coupent en $O$. On donne $\longueur{OA} = \qty{3}{cm}$, $\longueur{OB} = \qty{4}{cm}$, $\longueur{OC} = \qty{5}{cm}$, $\longueur{OD} = \qty{4}{cm}$ et $\widehat{AOB} = \ang{50}$.
    
    \item Construis un quadrilatère $ABCD$ tel que $\longueur{AB} = \qty{8}{cm}$, $\longueur{BC} = \qty{5}{cm}$. Place le point $E$ milieu de $[AB]$ et le point $F$ milieu de $[CD]$. Construis le quadrilatère de sorte que $\longueur{EF} = \qty{6}{cm}$ et que $EF \perp AB$.
    
    \item Construis un carré dont la diagonale mesure \qty{7}{cm}.
    
    \item Construis un quadrilatère croisé $ABCD$ avec $\longueur{AB} = \qty{4}{cm}$, $\longueur{BC} = \qty{5}{cm}$, $\longueur{CD} = \qty{4}{cm}$, $\longueur{DA} = \qty{5}{cm}$ et $\widehat{ABC} = \ang{60}$.
\end{enumerate}

\section*{Attendus}
\begin{enumerate}[label=\textbf{\arabic*.}]
    \item Identifier des figures simples (triangle, quadrilatère) sur base de caractéristiques : les côtés (nombre, longueur, parallélisme, perpendicularité) ; les angles (amplitude).
    \item Construire un losange connaissant la longueur du côté et l'amplitude d'un angle.
    \item Construire un parallélogramme connaissant la longueur de deux côtés et l'amplitude de l'angle compris entre eux.
    \item Énoncer la propriété de la somme des amplitudes des angles intérieurs d'un triangle et d'un quadrilatère.
    \item Énoncer les propriétés relatives aux angles, aux axes et centre de symétrie, aux droites remarquables, dans des figures simples.
    \item Déterminer l'amplitude d'un angle en utilisant la propriété relative à la somme des amplitudes des angles intérieurs d'un triangle et d'un quadrilatère.
    \item Terminer la construction d'une figure simple respectant des contraintes sur les droites remarquables.
    \item Construire une figure simple dont le centre de symétrie, le(s) axe(s) de symétrie sont donnés.
    \item Construire un quadrilatère (carré, rectangle, losange, parallélogramme) dont une diagonale ou une médiane est donnée.
\end{enumerate}
